prowsetk_add_library(prowsetk_automation support.cpp)
target_link_libraries(prowsetk_automation PUBLIC ProwseTk::core ProwseTk::tomlplusplus)
target_include_directories(prowsetk_automation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

# Embed modules so installed executables have the same Lua implementation as
# build-tree tools, without source-tree paths or a second implementation.
set(_lua_header "#pragma once\nnamespace prowsetk::automation {\n")
foreach(_module ezlogin scrape_endpoints lcrawler lpgwatch)
    if(_module STREQUAL "ezlogin")
        set(_source "${PROJECT_SOURCE_DIR}/plugins/ezlogin/lua/ezlogin.lua")
    elseif(_module STREQUAL "scrape_endpoints")
        set(_source "${PROJECT_SOURCE_DIR}/plugins/scrape-endpoints/lua/scrape_endpoints.lua")
    elseif(_module STREQUAL "lcrawler")
        set(_source "${PROJECT_SOURCE_DIR}/tools/crawler/lua/lcrawler.lua")
    else()
        set(_source "${PROJECT_SOURCE_DIR}/tools/pagewatch/lua/lpgwatch.lua")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source}")
    file(READ "${_source}" _text)
    string(APPEND _lua_header "inline constexpr const char* ${_module}_source = R\"PWTK_LUA(${_text})PWTK_LUA\";\n")
endforeach()
string(APPEND _lua_header "}\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lua_sources.hpp" CONTENT "${_lua_header}" @ONLY)
target_include_directories(prowsetk_automation PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
