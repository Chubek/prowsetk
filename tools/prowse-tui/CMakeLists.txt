option(PROWSETK_BUILD_TUI "Build the Termscript terminal browser" ON)
if(PROWSETK_BUILD_TUI AND TARGET ProwseTk::termlib AND TARGET ProwseTk::termscript)
    prowsetk_add_library(prowse_tui_model render.cpp controller.cpp help.cpp settings.cpp)
    target_link_libraries(prowse_tui_model PUBLIC ProwseTk::core)
    target_include_directories(prowse_tui_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    if(TARGET ProwseTk::tomlplusplus)
        target_link_libraries(prowse_tui_model PRIVATE ProwseTk::tomlplusplus)
        target_compile_definitions(prowse_tui_model PRIVATE PROWSETK_TUI_TOML=1)
    endif()
    # Embed from the man sources so help works in both build and installed binaries.
    set(_help_pages overview commands search config extensions)
    set(_help_header "#pragma once\nstruct HelpSource { const char* name; const char* source; };\ninline constexpr HelpSource help_sources[] = {\n")
    foreach(_page IN LISTS _help_pages)
        set(_source "${CMAKE_CURRENT_SOURCE_DIR}/help/${_page}.tsh")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source}")
        file(READ "${_source}" _text)
        string(APPEND _help_header "{\"${_page}\", R\"PWTK_MAN(${_text})PWTK_MAN\"},\n")
        install(FILES "${_source}" DESTINATION "${CMAKE_INSTALL_DATADIR}/prowse-tui/help")
    endforeach()
    string(APPEND _help_header "};\n")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/help_pages.hpp" CONTENT "${_help_header}" @ONLY)
    target_include_directories(prowse_tui_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    add_library(prowse_tui_terminal STATIC terminal.c termscript_ui.c)
    target_include_directories(prowse_tui_terminal PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(prowse_tui_terminal PRIVATE PWTK_UI_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/ui.ts")
    target_link_libraries(prowse_tui_terminal PRIVATE ProwseTk::termlib ProwseTk::termscript)
    prowsetk_add_executable(prowse-tui main.cpp)
    target_link_libraries(prowse-tui PRIVATE prowse_tui_model prowse_tui_terminal ProwseTk::termlib ProwseTk::termscript)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/ui.ts" DESTINATION "${CMAKE_INSTALL_DATADIR}/prowse-tui")
    install(TARGETS prowse-tui RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()
