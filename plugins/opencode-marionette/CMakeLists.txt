# Reuses host transport, bounded JSON bridge and existing extraction plugins.
add_library(prowsetk_opencode_marionette_static STATIC src/decisions.cpp src/marionette.cpp)
target_include_directories(prowsetk_opencode_marionette_static PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(prowsetk_opencode_marionette_static PUBLIC prowsetk_opencode_bridge_static prowsetk_scrape_endpoints_static prowsetk_schema_grabber_static)
set_target_properties(prowsetk_opencode_marionette_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
prowsetk_set_warnings(prowsetk_opencode_marionette_static)
add_library(prowsetk_opencode_marionette SHARED src/plugin_entry.cpp src/decisions.cpp src/marionette.cpp)
target_include_directories(prowsetk_opencode_marionette PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(prowsetk_opencode_marionette PUBLIC prowsetk_opencode_bridge_static prowsetk_scrape_endpoints_static prowsetk_schema_grabber_static)
prowsetk_set_warnings(prowsetk_opencode_marionette)
if(PROWSETK_BUILD_CLI)
    prowsetk_add_executable(ptk_opencode_marionette src/main.cpp)
    target_link_libraries(ptk_opencode_marionette PRIVATE prowsetk_opencode_marionette_static)
    set_target_properties(ptk_opencode_marionette PROPERTIES OUTPUT_NAME ptk-opencode-marionette)
    install(TARGETS ptk_opencode_marionette RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()
install(TARGETS prowsetk_opencode_marionette LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/prowsetk/plugins")
install(FILES include/prowsetk/plugins/opencode_marionette.hpp DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/prowsetk/plugins")
install(FILES README.md decisions.example.json DESTINATION "${CMAKE_INSTALL_DATADIR}/prowsetk/plugins/opencode-marionette")
