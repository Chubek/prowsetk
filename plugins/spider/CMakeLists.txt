option(PROWSETK_BUILD_SPIDER "Build persistent spider plugin and daemons" ON)
if(NOT PROWSETK_BUILD_SPIDER OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT TARGET ProwseTk::lmdbxx)
    message(STATUS "spider: disabled (requires Linux, LMDB and lmdbxx)")
    return()
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lua/lspider.lua" SPIDER_LUA_SOURCE)
configure_file(src/lspider_source.hpp.in generated/lspider_source.hpp @ONLY)
prowsetk_add_library(prowsetk_spider_static src/cache.cpp src/protocol.cpp src/spider.cpp)
target_include_directories(prowsetk_spider_static PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(prowsetk_spider_static PUBLIC ProwseTk::core PRIVATE ProwseTk::lmdbxx)
add_library(prowsetk_spider SHARED src/plugin_entry.cpp)
target_link_libraries(prowsetk_spider PRIVATE prowsetk_spider_static)
prowsetk_set_warnings(prowsetk_spider)
prowsetk_add_executable(ptkspiderd src/daemon.cpp)
target_link_libraries(ptkspiderd PRIVATE prowsetk_spider_static)
if(CMAKE_CXX_FLAGS MATCHES "fsanitize=address")
    target_compile_definitions(ptkspiderd PRIVATE SPIDER_SANITIZED=1)
endif()
prowsetk_add_executable(ptkspiderctl src/controller.cpp)
target_link_libraries(ptkspiderctl PRIVATE prowsetk_spider_static)
install(TARGETS prowsetk_spider LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/prowsetk/plugins")
install(TARGETS ptkspiderd ptkspiderctl RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES include/prowsetk/plugins/spider.hpp
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/prowsetk/plugins")
install(FILES lua/lspider.lua DESTINATION "${CMAKE_INSTALL_DATADIR}/prowsetk/plugins/spider")
install(FILES ptkspider.service README.md
    DESTINATION "${CMAKE_INSTALL_DATADIR}/prowsetk/plugins/spider")
