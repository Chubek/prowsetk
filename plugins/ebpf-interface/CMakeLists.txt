if(NOT TARGET ProwseTk::core)
  message(STATUS "ebpf-interface: ProwseTk::core not available, skipping")
  return()
endif()
add_library(prowsetk_ebpf_interface_static STATIC src/ebpf_interface.cpp)
target_include_directories(prowsetk_ebpf_interface_static PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(prowsetk_ebpf_interface_static PUBLIC ProwseTk::core)
set_target_properties(prowsetk_ebpf_interface_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(PROWSETK_HAVE_LIBBPF)
  target_compile_definitions(prowsetk_ebpf_interface_static PRIVATE PROWSETK_HAVE_LIBBPF=1)
  target_link_libraries(prowsetk_ebpf_interface_static PUBLIC ProwseTk::libbpf)
endif()
add_library(prowsetk_ebpf_interface SHARED src/ebpf_interface.cpp src/plugin_entry.cpp)
target_include_directories(prowsetk_ebpf_interface PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(prowsetk_ebpf_interface PRIVATE ProwseTk::core)
if(PROWSETK_HAVE_LIBBPF)
  target_compile_definitions(prowsetk_ebpf_interface PRIVATE PROWSETK_HAVE_LIBBPF=1)
  target_link_libraries(prowsetk_ebpf_interface PRIVATE ProwseTk::libbpf)
endif()
set_target_properties(prowsetk_ebpf_interface PROPERTIES OUTPUT_NAME "prowsetk_ebpf_interface" PREFIX "lib")
prowsetk_set_warnings(prowsetk_ebpf_interface)
prowsetk_set_warnings(prowsetk_ebpf_interface_static)
install(TARGETS prowsetk_ebpf_interface LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/prowsetk/plugins")
install(FILES lua/ebpf_interface.lua DESTINATION "${CMAKE_INSTALL_DATADIR}/prowsetk/plugins/ebpf-interface")
install(FILES include/prowsetk/plugins/ebpf_interface.hpp DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/prowsetk/plugins")
if(PROWSETK_BUILD_TESTS AND TARGET GTest::gtest_main)
  include(GoogleTest)
  add_executable(prowsetk_plugin_ebpf_interface tests/test_ebpf_interface.cpp)
  target_link_libraries(prowsetk_plugin_ebpf_interface PRIVATE prowsetk_ebpf_interface_static GTest::gtest_main)
  if(PROWSETK_HAVE_LIBBPF)
    target_compile_definitions(prowsetk_plugin_ebpf_interface PRIVATE PROWSETK_HAVE_LIBBPF=1)
  endif()
  prowsetk_set_warnings(prowsetk_plugin_ebpf_interface)
  gtest_discover_tests(prowsetk_plugin_ebpf_interface
    TEST_PREFIX "prowsetk.plugin.ebpf-interface."
    PROPERTIES LABELS "unit;plugin;network" TIMEOUT 30
    DISCOVERY_MODE PRE_TEST DISCOVERY_TIMEOUT 30)
endif()
